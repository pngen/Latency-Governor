enable_testing()

function(lg_add_test name src)
    add_executable(${name} ${src})
    target_link_libraries(${name} PRIVATE latency_governor latency_governor_protocol latency_governor_backends)
    if(LATENCY_GOVERNOR_ENABLE_CUDA)
        target_link_libraries(${name} PRIVATE latency_governor_backends_cuda)
    endif()
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lg_apply_warnings(${name})
    add_test(NAME ${name} COMMAND ${name})
endfunction()

lg_add_test(test_core test_core.cpp)
lg_add_test(test_property test_property.cpp)
lg_add_test(test_persistence test_persistence.cpp)
lg_add_test(test_protocol test_protocol.cpp)
lg_add_test(test_concurrency test_concurrency.cpp)
lg_add_test(test_cuda test_cuda.cpp)
lg_add_test(test_multiprocess_fencing test_multiprocess_fencing.cpp)
